	\section*{Task 2: Project Objectives \& Risk Management}
	
	\subsection*{1. Project Objectives (Mục tiêu Dự án)}
	
	\begin{itemize}
		\item \textbf{Timeline Objective (Tiến độ):} Hoàn thành toàn bộ quy trình phát triển, kiểm thử và nghiệm thu hệ thống Cruise Management System trong vòng \textbf{8 tuần}.
		\item \textbf{Effort Objective (Công sức):} Kiểm soát và tối ưu tổng công sức thực hiện dự án trong mức giới hạn \textbf{40 Man-days}.
		\item \textbf{Defect Distribution Target (Phân bố chỉ tiêu phát hiện lỗi):}
		\begin{itemize}
			\item \textbf{Giai đoạn Internal / Unit Testing:} Phát hiện và xử lý tối thiểu \textbf{70\%} tổng số lỗi hệ thống.
			\item \textbf{Giai đoạn System Integration Testing:} Phát hiện và xử lý khoảng \textbf{20\%} số lỗi.
			\item \textbf{Giai đoạn User Acceptance Testing (UAT):} Kiểm soát số lỗi phát hiện còn lại dưới \textbf{10\%}.
			\item \textbf{Môi trường Production (Sau khi ra mắt):} \textbf{0\%} lỗi nghiêm trọng (Critical / Blocker).
		\end{itemize}
	\end{itemize}
