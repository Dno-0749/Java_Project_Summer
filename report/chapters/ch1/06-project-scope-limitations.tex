\section{Phạm vi và giới hạn của dự án}
\subsection{Các tính năng chính}

\subsubsection{Tính năng dành cho hành khách}
\begin{itemize}
    \item \textbf{FE-01:} Xem lịch trình chuyến đi, các hoạt động giải trí trên tàu và các chuyến tham quan bờ.
    \item \textbf{FE-02:} Đăng ký trước các hoạt động hoặc dịch vụ đặc biệt, hoặc đăng ký trực tiếp nếu còn chỗ.
    \item \textbf{FE-03:} Phân biệt dịch vụ miễn phí và trả phí.
    \item \textbf{FE-04:} Xác thực check-in tham gia hoạt động bằng mã QR, thẻ du thuyền hoặc vòng tay RFID/NFC.
    \item \textbf{FE-05:} Xem chi phí đã phát sinh và theo dõi hóa đơn tạm tính.
    \item \textbf{FE-06:} Nhận thông báo về thay đổi lịch trình, thời gian tập trung và thời gian quay trở lại tàu.
    \item \textbf{FE-07:} Gửi phản hồi sau khi hoàn thành hoạt động hoặc cuối chuyến đi.
\end{itemize}

\subsubsection{Tính năng dành cho bộ phận điều phối lịch trình}
\begin{itemize}
    \item \textbf{FE-01:} Tạo và quản lý các chuyến du lịch nhiều ngày và lịch trình chi tiết mỗi ngày.
    \item \textbf{FE-02:} Quản lý các cảng ghé thăm và thiết lập thời gian tàu cập bến hoặc rời bến.
    \item \textbf{FE-03:} Cập nhật thay đổi lịch trình do thời tiết hoặc vận hành và thông báo đến hành khách và các bộ phận liên quan.
\end{itemize}

\subsubsection{Tính năng dành cho bộ phận quản lý hoạt động trên tàu}
\begin{itemize}
    \item \textbf{FE-01:} Tạo các hoạt động giải trí và cấu hình thời gian, địa điểm, sức chứa, chi phí.
    \item \textbf{FE-02:} Quản lý danh sách đăng ký, giám sát tỷ lệ tham gia và phản hồi.
    \item \textbf{FE-03:} Kiểm tra check-in hành khách bằng các phương thức xác thực.
\end{itemize}

\subsubsection{Tính năng dành cho bộ phận quản lý tham quan bờ}
\begin{itemize}
    \item \textbf{FE-01:} Tạo các chuyến tham quan, quản lý đơn vị cung cấp, thiết lập sức chứa, giá và thời gian tập trung.
    \item \textbf{FE-02:} Quản lý danh sách hành khách theo nhóm, phương tiện vận chuyển hoặc hướng dẫn viên.
    \item \textbf{FE-03:} Theo dõi trạng thái chuyến tham quan đã hoàn thành, hủy hoặc trì hoãn.
\end{itemize}

\subsection{Giới hạn của dự án}
Dự án tập trung vào các chức năng cốt lõi của hệ thống, chưa triển khai đầy đủ các tính năng nâng cao như tích hợp thanh toán quốc tế, phân tích dữ liệu lớn hoặc kết nối với nhiều nền tảng bên ngoài.
 \subsubsection{Tính năng dành cho Nhân viên bán hàng và dịch vụ trên tàu} \begin{itemize} \item \textbf{FE-01:} Ghi nhận dịch vụ hoặc sản phẩm sử dụng, xác thực hành khách qua các thiết bị. \item \textbf{FE-02:} Ghi nợ trực tiếp vào tài khoản trên tàu của hành khách, kiểm tra dịch vụ trong gói. \item \textbf{FE-03:} Thực hiện hủy giao dịch hoặc hoàn tiền (nếu được cấp quyền). \item \textbf{FE-04:} Lưu trữ tạm thời giao dịch khi mạng không ổn định và đồng bộ hóa sau. \end{itemize} \subsubsection{Tính năng dành cho Bộ phận Tài chính và Lễ tân} \begin{itemize} \item \textbf{FE-01:} Quản lý tài khoản trên tàu theo hành khách, phòng hoặc mã đặt chỗ. \item \textbf{FE-02:} Đối soát giao dịch POS, giải quyết tranh chấp chi phí và thực hiện thanh toán cuối cùng. \item \textbf{FE-03:} Xuất hóa đơn cuối cùng khi kết thúc chuyến đi. \end{itemize} \subsubsection{Tính năng dành cho Quản lý vận hành} \begin{itemize} \item \textbf{FE-01:} Theo dõi bảng điều khiển (dashboard) về hoạt động, dịch vụ, doanh thu và sức chứa. \item \textbf{FE-02:} Giám sát trạng thái check-in và quản lý rủi ro hành khách quay lại tàu muộn. \item \textbf{FE-03:} Xem báo cáo vận hành sau chuyến đi. \end{itemize} \subsubsection{Tính năng dành cho Quản trị viên hệ thống} \begin{itemize} \item \textbf{FE-01:} Quản lý tài khoản người dùng, phân quyền, quản lý tàu, khu vực và điểm dừng. \item \textbf{FE-02:} Cấu hình chính sách hệ thống (đăng ký, hủy, hoàn tiền). \item \textbf{FE-03:} Quản lý mã QR, thẻ, thiết bị RFID/NFC và xử lý log hệ thống. \end{itemize} \subsection{Hạn chế và loại trừ} \begin{itemize} \item \textbf{LI-1: Sự phụ thuộc vào phần cứng:} Hệ thống dựa vào các thiết bị POS vật lý, máy đọc QR/RFID/NFC và các loại thẻ/vòng tay chuyên dụng. \item \textbf{LI-2: Độ trễ đồng bộ mạng:} Do giao dịch được lưu trữ tạm thời trong lúc mạng không ổn định, bảng điều khiển trực tiếp của bộ phận Vận hành và Tài chính có thể không phản ánh dữ liệu thời gian thực 100\% cho đến khi đồng bộ xong. \item \textbf{LI-3: Chức năng định vị BLE:} Tính năng theo dõi vị trí hành khách bằng thiết bị đeo BLE chỉ là module tùy chọn. Nếu áp dụng, nó chỉ ước tính vị trí hành khách ở mức độ ``khu vực'' thay vì độ chính xác tuyệt đối. \end{itemize}
