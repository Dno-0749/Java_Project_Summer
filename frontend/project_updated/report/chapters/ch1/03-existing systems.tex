\section{Hệ thống hiện tại}
\subsection{Giải pháp quản lý du thuyền truyền thống}
\textbf{Mô tả:} Nhiều hệ thống truyền thống dựa trên các phần mềm rời rạc cho từng bộ phận, ví dụ như hệ thống đặt chỗ riêng và quản lý bán hàng riêng, kết hợp với vé giấy.

\textbf{Nhược điểm:}
\begin{itemize}
    \item Thiếu khả năng đồng bộ ngoại tuyến khi mất kết nối mạng, dẫn đến gián đoạn quy trình tại các điểm bán hàng (POS).
    \item Không cung cấp cho hành khách cái nhìn tổng thể theo thời gian thực về chi phí đã phát sinh và hóa đơn tạm tính.
    \item Sự phối hợp giữa các đơn vị cung cấp dịch vụ tham quan và bộ phận quản lý lịch trình trên tàu còn kém hiệu quả.
\end{itemize}

\subsection{Hệ thống POS thông thường}
\textbf{Mô tả:} Các hệ thống POS bán lẻ tiêu chuẩn dùng trong nhà hàng hoặc cửa hàng trên tàu.

\textbf{Nhược điểm:}
\begin{itemize}
    \item Không tích hợp với tài khoản chi tiêu trung tâm hoặc thông tin đặt chỗ của hành khách trên tàu.
    \item Không hỗ trợ các thiết bị đặc thù như thẻ du thuyền hoặc vòng tay RFID/NFC để xác thực thanh toán không dùng tiền mặt.
\end{itemize}
